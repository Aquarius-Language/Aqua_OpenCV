cmake_minimum_required(VERSION 3.20)
project(Aqua_OpenCV LANGUAGES CXX)
find_package(OpenCV 4.12 REQUIRED COMPONENTS core imgproc imgcodecs features2d calib3d video videoio objdetect photo stitching dnn)
add_library(aqua_opencv SHARED native/opencv_bridge.cpp)
target_compile_features(aqua_opencv PRIVATE cxx_std_17)
target_include_directories(aqua_opencv PRIVATE vendor ${OpenCV_INCLUDE_DIRS})
target_link_libraries(aqua_opencv PRIVATE ${OpenCV_LIBS})
if(MSVC)
  target_compile_options(aqua_opencv PRIVATE /O2 /EHsc)
else()
  target_compile_options(aqua_opencv PRIVATE -O3 -fvisibility=hidden)
endif()
if(APPLE)
  set_target_properties(aqua_opencv PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(aqua_opencv PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS aqua_opencv RUNTIME DESTINATION . LIBRARY DESTINATION .)
